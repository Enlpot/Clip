cmake_minimum_required(VERSION 4.2)
project(clip_macos_fixture VERSION 1.0.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 15.0)
set(CLIP_SHOT_VERSION 1.0.5-beta)
set(CLIP_SHOT_VERSION_NUMERIC 1.0.5)
set(CLIP_SHOT_VENDOR "Clip Apps")
set(CLIP_SHOT_COPYRIGHT "Copyright (C) 2025-2026 mg-chao")
# This fixture provisions the shared Qt/ONNX runtime deployment path.
set(CLIP_SHOT_RELEASE_STATIC OFF)
set(CLIP_SHOT_QT_STATIC OFF)
set(CLIP_SHOT_OCR_STATIC_ONNXRUNTIME OFF)
if(NOT CLIP_FFMPEG_ROOT)
    message(FATAL_ERROR "Pass CLIP_FFMPEG_ROOT to the provisioned native FFmpeg prefix")
endif()
list(APPEND CMAKE_MODULE_PATH "${CLIP_ROOT}/cmake")
include(ClipQt)
clip_find_qt(COMPONENTS Widgets)
add_library(clip_capture_c INTERFACE)
add_library(clip_recording_c INTERFACE)
add_library(onnxruntime SHARED dummy.cpp)
add_library(onnxruntime::onnxruntime ALIAS onnxruntime)
set_target_properties(onnxruntime PROPERTIES VERSION 1.28.0 SOVERSION 1)
add_library(clip_shot_image_codec_backend SHARED dummy.cpp)
add_executable(clip_shot MACOSX_BUNDLE main.cpp)
target_link_libraries(clip_shot PRIVATE Qt6::Widgets clip_recording_c clip_shot_image_codec_backend)
add_dependencies(clip_shot onnxruntime)
add_executable(clip-ocr-process helper.cpp)
set(CMAKE_INSTALL_DEFAULT_COMPONENT_NAME ClipShot)
install(TARGETS clip_shot BUNDLE DESTINATION .)
install(TARGETS clip_shot_image_codec_backend
    LIBRARY DESTINATION clip_shot.app/Contents/Frameworks)
install(TARGETS clip-ocr-process RUNTIME DESTINATION clip_shot.app/Contents/MacOS)
find_program(CLIP_CRASHPAD_HANDLER NAMES crashpad_handler
    PATHS "${CLIP_FFMPEG_ROOT}/tools/crashpad" "${CLIP_FFMPEG_ROOT}/tools"
    NO_DEFAULT_PATH REQUIRED)
find_library(CLIP_FIXTURE_ZLIB NAMES z PATHS "${CLIP_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
if(NOT TARGET ZLIB::ZLIB)
    add_library(ZLIB::ZLIB UNKNOWN IMPORTED)
    set_target_properties(ZLIB::ZLIB PROPERTIES IMPORTED_LOCATION "${CLIP_FIXTURE_ZLIB}")
endif()
include(ClipShotMacOS)
target_link_libraries(onnxruntime PRIVATE PkgConfig::CLIP_SHOT_FFMPEG)
target_link_libraries(clip_shot_image_codec_backend PRIVATE PkgConfig::CLIP_SHOT_FFMPEG)
include(ClipShotMacOSPackage)
