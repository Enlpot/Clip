cmake_minimum_required(VERSION 3.25)
project(ClipQtTimeZoneContract LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/ClipQt.cmake")
clip_find_qt(COMPONENTS Core)
add_executable(clip-qt-timezone-tests timezone_tests.cpp)
target_link_libraries(clip-qt-timezone-tests PRIVATE Qt6::Core)
enable_testing()
add_test(NAME clip-qt-timezone-tests COMMAND clip-qt-timezone-tests)
set_tests_properties(clip-qt-timezone-tests PROPERTIES LABELS unit)
