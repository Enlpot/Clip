cmake_minimum_required(VERSION 3.27)
project(clip_media_native LANGUAGES C CXX)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
option(CLIP_MEDIA_BUILD_RECORDING "Build FFmpeg recording/export C library" OFF)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/RustStaticLibrary.cmake")
clip_add_rust_static_library(clip_capture_c PACKAGE clip-capture-c
    MANIFEST_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
target_include_directories(clip_capture_c INTERFACE "${CMAKE_CURRENT_LIST_DIR}/../crates/clip-capture-c/include")
if(APPLE)
    foreach(_framework AppKit CoreText Carbon CloudKit CoreData Foundation CoreFoundation CoreGraphics CoreMedia CoreVideo
        ScreenCaptureKit AVFoundation AVFAudio AudioToolbox CoreAudio Metal CoreImage
        QuartzCore IOSurface ImageIO UniformTypeIdentifiers MediaToolbox)
        target_link_libraries(clip_capture_c INTERFACE "-framework ${_framework}")
    endforeach()
    target_link_libraries(clip_capture_c INTERFACE objc iconv c++)
    add_executable(clip_capture_c_smoke c_capture_smoke.c)
    target_link_libraries(clip_capture_c_smoke PRIVATE clip_capture_c)
    enable_testing()
    add_test(NAME clip-capture-c-macos-abi COMMAND clip_capture_c_smoke)
endif()
if(CLIP_MEDIA_BUILD_RECORDING)
    clip_add_rust_static_library(clip_recording_c PACKAGE clip-recording-c
        MANIFEST_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
    target_include_directories(clip_recording_c INTERFACE "${CMAKE_CURRENT_LIST_DIR}/../crates/clip-recording-c/include")
    find_package(PkgConfig REQUIRED)
    if(DEFINED ENV{FFMPEG_DIR})
        set(ENV{PKG_CONFIG_PATH} "$ENV{FFMPEG_DIR}/lib/pkgconfig")
    endif()
    pkg_check_modules(CLIP_MEDIA_FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale libswresample)
    target_link_libraries(clip_recording_c INTERFACE clip_capture_c PkgConfig::CLIP_MEDIA_FFMPEG)
    if(APPLE)
        add_executable(clip_recording_c_smoke c_recording_smoke.c)
        target_link_libraries(clip_recording_c_smoke PRIVATE clip_recording_c)
        add_test(NAME clip-recording-c-macos-abi COMMAND clip_recording_c_smoke)
    endif()
endif()
